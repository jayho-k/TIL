## 2-1)  A2A vs MCP

- MCP
  - Tool, DB 등 외부 자원을 사용하는 것을 목표로 한다. 
  - 도
- A2A
  - 다른 Agent가 다른 Sub Agent 들을 지원받기 위해서 사용하는 것
  - 즉 이때 호출되는 Agent의 Tool이나, MCP 등을 사용하는 것은 필요없다. 
  - 즉 계층이 다르다고 책임이 다르다.
    - 계획 대화 상태 등을 책임지는 건 그쪽 Agent 쪽이라고 보면 된다.
    - 하지만 MCP로 진행했을 때 도구 사용법만 확인하기 때문에 Agent 는 직접 개발해야한다는 뜻이 된다.
    - 예를 들어서 사내 게시판을 호출할 때 Search tool만 MCP로 열어놨고, 이후 게시판 Agent를 따로 만들어서 그것을 가져와 사용하는 방법을 사용했다. 만약 A2A를 사용해서 그쪽에서 Agent를 만들었다면 그냥 그것을 사용하면 됐을 거 같다. 

- A2S의 단점
  - 설계가 복잡해짐
  - 내가 원하는 Agent의 로직이 아닐 수 있음 >> 이건 뭐 어쩔수 없지 >> 내 서비스 관리하는 거니깐

### 어떤 값이 들어가야하지? 

```python
import os
import sys

import httpx

CARD_REQUIRED = [
    "name", "description", "supportedInterfaces", "version",
    "capabilities", "defaultInputModes", "defaultOutputModes", "skills",
]
SKILL_REQUIRED = ["id", "name", "description", "tags"]
INTERFACE_REQUIRED = ["url", "protocolBinding", "protocolVersion"]

url = os.environ["WORKER_URL"] + "/.well-known/agent-card.json"
card = httpx.get(url, timeout=30).json()

problems = []

for field in CARD_REQUIRED:
    if field not in card:
        problems.append(f"카드에 {field} 없음")

for index, interface in enumerate(card.get("supportedInterfaces", [])):
    for field in INTERFACE_REQUIRED:
        if field not in interface:
            problems.append(f"supportedInterfaces[{index}]에 {field} 없음")

ids = []
for index, skill in enumerate(card.get("skills", [])):
    for field in SKILL_REQUIRED:
        if field not in skill or not skill[field]:
            problems.append(f"skills[{index}]에 {field} 없거나 비어 있음")
    ids.append(skill.get("id"))

if len(ids) != len(set(ids)):
    problems.append("skills의 id가 겹침")

if problems:
    print("문제 있음:")
    for problem in problems:
        print(" -", problem)
    sys.exit(1)

print("필수 필드 다 있음")
print("skill id:", ids)
```

## Adapter (Lang Graph)

- Graph <> Adapter <> A2S Server

```python
import os

from langgraph.checkpoint.postgres import PostgresSaver
from langgraph.types import Command

from graph import build_graph

_saver_cm = None
_graph = None

def get_graph():
    """서버가 실행되는 동안 그래프와 체크포인터 연결을 유지한다."""
    global _saver_cm, _graph
    if _graph is None:
        _saver_cm = PostgresSaver.from_conn_string(os.environ["POSTGRES_URL"])
        saver = _saver_cm.__enter__()
        saver.setup()
        _graph = build_graph(checkpointer=saver)
    return _graph

def thread_of(task_id: str) -> dict:
    """외부 taskId로 내부 thread_id 설정을 만든다."""
    return {"configurable": {"thread_id": f"task-{task_id}"}}

def start(task_id: str, trip_id: str) -> dict:
    graph = get_graph()
    graph.invoke({"trip_id": trip_id, "log": []}, thread_of(task_id))
    return read(task_id)

def resume(task_id: str, answer: str) -> dict:
    graph = get_graph()
    graph.invoke(Command(resume=answer), thread_of(task_id))
    return read(task_id)

def read(task_id: str) -> dict:
    """LangGraph 상태에서 외부로 전달할 값만 골라낸다."""
    snapshot = get_graph().get_state(thread_of(task_id))
    values = snapshot.values
    waiting = "escalate" in (snapshot.next or ())
    # step 4 -> 다음 노드 ('escalate')
    return {
        "waiting": waiting,
        # "question": (snapshot.interrupts[0].value.get("question")
        #              if waiting and snapshot.interrupts else None),
        "ask": (snapshot.interrupts[0].value
                if waiting and snapshot.interrupts else None),
        "result": {
            "trip_id": values.get("trip_id"),
            "total": values.get("total"),
            "violations": values.get("violations") or [],
            "receipts": len(values.get("receipts") or []),
        },
        "approved": values.get("approved"),
    }

def start_stream(task_id: str, trip_id: str):
    """노드가 끝날 때마다 하나씩 내보낸다. stream_mode="updates"를 그대로 쓴다."""
    graph = get_graph()
    for chunk in graph.stream({"trip_id": trip_id, "log": []},
                              thread_of(task_id), stream_mode="updates"):
        for node, update in chunk.items():
            if node == "__interrupt__":
                continue
            yield node, update
```

# receiver
from fastapi import FastAPI, Request
from fastapi.responses import JSONResponse

app = FastAPI()

# 웹훅을 등록할 때 정해서 에이전트에 넘긴 값들
EXPECTED_CRED = "hook-cred-9"          # 에이전트가 자신을 증명할 때 쓴다
TASK_TOKENS: dict[str, str] = {}       # taskId → 이 태스크용 token

RECEIVED: list[dict] = []
REJECTED: list[dict] = []


@app.post("/hook")
async def hook(request: Request):
    body = await request.json()
    event = body.get("statusUpdate") or {}
    task_id = event.get("taskId")

    auth = request.headers.get("authorization", "")
    if auth != f"Bearer {EXPECTED_CRED}":
        REJECTED.append({"why": "보낸 쪽을 못 믿음", "auth": auth})
        return JSONResponse({"error": "unauthorized"}, status_code=401)

    token = request.headers.get("x-a2a-notification-token")
    expected = TASK_TOKENS.get(task_id)
    if expected is None or token != expected:
        REJECTED.append({"why": "모르는 태스크이거나 token 불일치", "taskId": task_id})
        return JSONResponse({"error": "unknown task"}, status_code=400)

    RECEIVED.append({
        "taskId": task_id,
        "state": (event.get("status") or {}).get("state"),
    })
    return {"ok": True}


@app.post("/expect/{task_id}/{token}")
def expect(task_id: str, token: str):
    TASK_TOKENS[task_id] = token
    return {"ok": True}


@app.get("/received")
def received():
    return {"ok": RECEIVED, "rejected": REJECTED}

# push
import httpx

CONFIGS: dict[str, dict] = {}


def save(config: dict) -> dict:
    CONFIGS[config["taskId"]] = config
    return config


def get(task_id: str) -> dict | None:
    return CONFIGS.get(task_id)


def notify(task: dict) -> None:
    """태스크가 종료·중단 상태가 되면 호출한 쪽에 알린다."""
    config = CONFIGS.get(task["id"])
    if not config:
        return

    headers = {"Content-Type": "application/a2a+json"}
    auth = config.get("authentication") or {}
    if auth.get("scheme") and auth.get("credentials"):
        headers["Authorization"] = f"{auth['scheme']} {auth['credentials']}"
    if config.get("token"):
        headers["X-A2A-Notification-Token"] = config["token"]

    # 스트리밍과 같은 StreamResponse 모양이다. 상태 변화는 statusUpdate에 담는다.
    body = {"statusUpdate": {
        "taskId": task["id"],
        "contextId": task["contextId"],
        "status": task["status"],
    }}

    try:
        httpx.post(config["url"], json=body, headers=headers, timeout=10)
    except Exception:
        # 한 번만 보낸다. 실패해도 태스크는 그대로다.
        pass
